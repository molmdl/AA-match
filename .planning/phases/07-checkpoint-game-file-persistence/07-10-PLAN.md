```python
def peek_kind(path):
    """Read ONLY the container kind from an AA-match JSON file.

    The Phase-7 Import dispatch reads the kind to route 'game' vs
    'checkpoint' files through one button (header-exact -- never
    extension guesswork). Foreign/unparseable files raise the SAME
    FormatError classes as load_container (the refusal messages are
    shared verbatim); a structurally-valid container returns its
    kind string WITHOUT running any kind-specific validation.
    """
    raw = read_json_file(path)
    if not isinstance(raw, dict) or raw.get('magic') != AAM_MAGIC:
        raise FormatError(
            "not an AA-match file (magic=%r, expected %r)"
            % (raw.get('magic') if isinstance(raw, dict) else raw,
               AAM_MAGIC))
    return raw.get('kind')
```
